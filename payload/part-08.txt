(p.x * 4.0 + t * 1.7, p.y * 8.0 - t));
        shimmer *= 1.0 + pointerField * .28;

        vec3 green = vec3(0.21, 1.0, 0.56);
        vec3 cyan = vec3(0.20, 0.76, 1.0);
        vec3 aurora = mix(green, cyan, smoothstep(-.5, .9, p.x + n2 * .5));
        col += aurora * band * shimmer * .94;
        col += mix(cyan, green, .35) * veil * .30;

        col += vec3(0.03, 0.08, 0.095) * exp(-pow(p.y + .45, 2.0) * 6.0) * .48;
        float vignette = smoothstep(1.35, .25, length(p * vec2(.72, .92)));
        col *= .62 + .38 * vignette;

        vec2 starGrid = floor((uv + vec2(0.0, travel * .015)) * vec2(310.0, 190.0));
        float star = step(.9945, hash(starGrid));
        float upper = smoothstep(.36, .92, uv.y);
        col += vec3(star * upper * .58);

        gl_FragColor = vec4(col, 1.0);
      }
    `;

    function makeCanvasTexture() {
      const c = document.createElement('canvas');
      c.width = 2048; c.height = 1024;
      const ctx = c.getContext('2d');
      const ocean = ctx.createLinearGradient(0, 0, 0, c.height);
      ocean.addColorStop(0, '#163447');
      ocean.addColorStop(.45, '#0e2430');
      ocean.addColorStop(1, '#09131b');
      ctx.fillStyle = ocean;
      ctx.fillRect(0, 0, c.width, c.height);

      ctx.globalAlpha = .12;
      for (let i = 0; i < 5000; i++) {
        const x = Math.random() * c.width;
        const y = Math.random() * c.height;
        const s = Math.random() * 2 + .3;
        ctx.fillStyle = i % 3 ? '#9effdb' : '#7cc8ff';
        ctx.fillRect(x, y, s, s);
      }
      ctx.globalAlpha = 1;

      function blob(points, fill) {
        ctx.beginPath();
        ctx.moveTo(points[0][0], points[0][1]);
        for (let i = 1; i < points.length; i++) ctx.bezierCurveTo(...points[i]);
        ctx.closePath();
        ctx.fillStyle = fill;
        ctx.fill();
      }

      ctx.shadowColor = 'rgba(125, 247, 185, .18)';
      ctx.shadowBlur = 24;

      blob([[1390,292],[1440,252,1510,252,1576,286],[1624,314,1660,356,1665,402],[1670,450,1640,492,1585,514],[1512,545,1460,530,1420,498],[1382,465,1362,425,1364,382],[1366,338,1372,312,1390,292]], '#2c5a45');

      ctx.shadowBlur = 0;
      ctx.fillStyle = 'rgba(227, 247, 255, .72)';
      ctx.beginPath(); ctx.ellipse(1024, 108, 410, 70, 0, 0, Math.PI * 2); ctx.fill();
